# Stage 1: Build Rust Smart Contract & TypeScript Agent
FROM rust:1.79-bullseye AS builder

WORKDIR /usr/src/helios

# Install Solana and Node.js
RUN apt-get update && apt-get install -y pkg-config build-essential libudev-dev curl gnupg
RUN curl -fsSL https://deb.nodesource.com/setup_20.x | bash - && apt-get install -y nodejs
RUN sh -c "$(curl -sSfL https://release.anza.xyz/v1.18.26/installer)"
ENV PATH="/root/.local/share/solana/install/active_release/bin:${PATH}"
RUN cargo install --git https://github.com/coral-xyz/anchor --tag v0.30.1 anchor-cli --locked

COPY . .
RUN cd client && npm install && npm run build

# Stage 2: Minimalist Enterprise Runner
FROM node:20-alpine AS runner
WORKDIR /app
COPY --from=builder /usr/src/helios/client/dist ./dist
COPY --from=builder /usr/src/helios/client/node_modules ./node_modules
COPY --from=builder /usr/src/helios/client/package.json ./package.json

ENV NODE_ENV=production
HEALTHCHECK --interval=30s --timeout=5s --start-period=5s --retries=3 \
  CMD node -e "console.log('System Operational')" || exit 1

ENTRYPOINT ["node", "dist/agent.js"]
