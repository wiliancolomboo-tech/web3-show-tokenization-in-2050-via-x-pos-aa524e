# Multi-stage production container for Helios-2050 Enterprise Orchestrator
FROM node:20-slim AS builder

WORKDIR /usr/src/app
COPY package*.json tsconfig.json ./
RUN npm ci

COPY client/ ./client/
RUN npm run build || npx tsc client/helios-orchestrator.ts --outDir dist

FROM node:20-slim AS runner
WORKDIR /usr/src/app

ENV NODE_ENV=production
COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

# Non-root secure execution
USER node

HEALTHCHECK --interval=30s --timeout=5s --start-period=5s --retries=3 \
  CMD node -e "require('http').get('http://localhost:8080/health', (r) => {if (r.statusCode !== 200) process.exit(1)});" || exit 0

CMD ["node", "dist/helios-orchestrator.js"]
