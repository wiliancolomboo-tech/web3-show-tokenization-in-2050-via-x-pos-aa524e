# Multi-stage enterprise build for HELIOS-2050 Autonomous Daemon
FROM node:20-slim AS builder

WORKDIR /usr/src/app

COPY services/helios-orchestrator/package*.json ./
COPY services/helios-orchestrator/tsconfig.json ./
RUN npm ci

COPY services/helios-orchestrator/src ./src
RUN npm run build

# Runtime Stage
FROM node:20-slim AS runner

WORKDIR /usr/src/app
ENV NODE_ENV=production

COPY services/helios-orchestrator/package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

# Healthcheck checking process liveness
HEALTHCHECK --interval=30s --timeout=5s --start-period=5s --retries=3 \
  CMD node -e "process.exit(0)"

USER node
CMD ["node", "dist/index.js"]
